/*=============== 🟨 SHOW MENU ===============*/
const navMenu = document.getElementById('nav-menu'),
      navToggle = document.getElementById('nav-toggle'),
      navClose = document.getElementById('nav-close')

/* Show menu */
if(navToggle){
   navToggle.addEventListener('click', () =>{
      navMenu.classList.add('show-menu')
   })
}

/* Hide menu */
if(navClose){
   navClose.addEventListener('click', () =>{
      navMenu.classList.remove('show-menu')
   })
}


/*=============== 🟨 REMOVE MENU MOBILE ===============*/
const navLink = document.querySelectorAll('.nav__link')

const linkAction = () =>{
   const navMenu = document.getElementById('nav-menu')
   // When we click on each nav__link, we remove the show-menu class
   navMenu.classList.remove('show-menu')
}
navLink.forEach(n => n.addEventListener('click', linkAction))


/*=============== 🟨 CHANGE HEADER STYLES ===============*/
const scrollHeader = () =>{
   const header = document.getElementById('header')
   // Add the .scroll-header class if the bottom scroll of the viewport is greater than 50
   this.scrollY >= 50 ? header.classList.add('scroll-header') 
                      : header.classList.remove('scroll-header')
}
window.addEventListener('scroll', scrollHeader)


/*=============== 🟨 HOME SWIPER ANIMATION ===============*/
/* Add zeros to the left of the number (1 → 01 or 001) */
const formatNumber = (number, digits) => String(number).padStart(digits, '0')

/* Home number and home count of the active slide */
const homeNumber = document.getElementById('home-number'),
      homeCount = document.getElementById('home-count')

const updateNumbers = (swiper) =>{
   const current = swiper.realIndex + 1

   homeNumber.textContent = formatNumber(current, 2)
   homeCount.textContent = formatNumber(current, 3)
}

/* Roller effect: the old number goes up and the new one comes from below */
const animateNumbers = (swiper) =>{
   const numbersText = '.home__number-text, .home__count-text'

   // Stop the previous animation if the slide changes fast
   gsap.killTweensOf(numbersText)

   // opacity: 1 because gsap.defaults uses opacity: 0
   gsap.timeline({ defaults: { opacity: 1 } })
       .to(numbersText, {yPercent: -100, duration: 0.35, ease: 'power2.in',
                         onComplete: () => updateNumbers(swiper)})
       .set(numbersText, {yPercent: 100})
       .to(numbersText, {yPercent: 0, duration: 0.45, ease: 'power3.out'})
}

/* Add swiper slider */
const swiperHome = new Swiper('.home__swiper', {
   loop: true,
   speed: 800,
   effect: 'fade',
   fadeEffect: {
      crossFade: true,
   },
   parallax: true,
   grabCursor: true,
   keyboard: {
      enabled: true,
   },

   navigation: {
      nextEl: '.home__button-next',
      prevEl: '.home__button-prev',
   },
   autoplay: {
      delay: 4000,
      disableOnInteraction: false,
   },

   on: {
      init(){
         const total = this.slides.length

         // Each home card shows the number of the next slide
         this.slides.forEach(slide =>{
            const next = (Number(slide.dataset.swiperSlideIndex) + 1) % total + 1

            slide.querySelector('.home__card-number').textContent = formatNumber(next, 2)
         })

         updateNumbers(this)
      },
   },
})

/* Animate the numbers only after the slider starts (not on init) */
swiperHome.on('slideChange', () => animateNumbers(swiperHome))

/* Autoplay waits until the GSAP animation of the page ends */
swiperHome.autoplay.stop()

/* Next slide with the home card button */
const cardButtons = document.querySelectorAll('.home__card-button')

cardButtons.forEach(button => button.addEventListener('click', () =>{
   swiperHome.slideNext()
}))


/*=============== 🟨 GSAP ANIMATION ===============*/
gsap.defaults({ opacity: 0, ease: 'back.out(1.6)', duration: 1.4 })

const tl = gsap.timeline({ onComplete: () => swiperHome.autoplay.start() })

tl.from('.nav > *', {y: -30}, '.3')
  .from('.home__glow', {scale: 0}, '.6')
  .from('.home__img', {y: -200}, '.9')
  .from('.home__titles', {y: -200}, '1.3')
  .from('.home__number', {y: -30}, '1.6')
  .from('.home__card', {y: 120}, '1.8')
  .from('.home__button, .home__social, .home__count', {y: 120}, '2.1')
